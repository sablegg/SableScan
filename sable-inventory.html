<script>
// ── STORAGE ───────────────────────────────────
const INV_KEY = 'sable_inventory';
const loadInv  = () => { try { return JSON.parse(localStorage.getItem(INV_KEY)||'[]'); } catch(e){ return []; }};
const saveInv  = d => localStorage.setItem(INV_KEY, JSON.stringify(d));

// ── SEED DEMO ─────────────────────────────────
(function(){
  if (localStorage.getItem(INV_KEY)) return;
  const now = new Date();
  const d1 = new Date(now - 86400000*2).toISOString();
  const d2 = new Date(now - 86400000*7).toISOString();
  const seed = [
    // Session 1 — 2 days ago
    {id:1, name:'Charizard ex',   number:'199/165', cond:'NM', cost:38.40, ask:65.00, date:d1, session:'Market Run — Mar 14'},
    {id:2, name:'Pikachu VMAX',   number:'044/185', cond:'NM', cost:17.60, ask:28.00, date:d1, session:'Market Run — Mar 14'},
    {id:3, name:'Umbreon VMAX',   number:'215/264', cond:'LP', cost:24.32, ask:null,  date:d1, session:'Market Run — Mar 14'},
    // Session 2 — 7 days ago
    {id:4, name:'Rayquaza VSTAR', number:'111/189', cond:'NM', cost:12.00, ask:18.00, date:d2, session:'Card Kingdom'},
    {id:5, name:'Mewtwo V',       number:'030/198', cond:'MP', cost:4.50,  ask:null,  date:d2, session:'Card Kingdom'},
    {id:6, name:'Lugia V',        number:'186/195', cond:'NM', cost:22.00, ask:19.00, date:d2, session:'Card Kingdom'},
  ];
  saveInv(seed);
})();

// ── STATE ─────────────────────────────────────
let inventory = loadInv();
let openId = null;
let filter = 'all';
let nextId = Math.max(...inventory.map(c=>c.id), 0) + 1;

// ── CALCS ─────────────────────────────────────
function profit(c) { return c.ask != null ? c.ask - c.cost : null; }
function totalCost()   { return inventory.reduce((s,c)=>s+c.cost, 0); }
function totalAsk()    { return inventory.reduce((s,c)=>s+(c.ask||0), 0); }
function totalProfit() { return inventory.filter(c=>c.ask!=null).reduce((s,c)=>s+(c.ask-c.cost),0); }
function pricedCount() { return inventory.filter(c=>c.ask!=null).length; }
function marginPct()   {
  const cost=inventory.filter(c=>c.ask!=null).reduce((s,c)=>s+c.cost,0);
  const ask=inventory.filter(c=>c.ask!=null).reduce((s,c)=>s+c.ask,0);
  return cost>0 ? ((ask-cost)/cost*100) : 0;
}

// ── FILTER ────────────────────────────────────
function setFilter(f, el) {
  filter = f;
  document.querySelectorAll('.filter-btn').forEach(b=>b.classList.remove('active'));
  el.classList.add('active');
  render();
}

function filteredItems() {
  if (filter==='unpriced') return inventory.filter(c=>c.ask==null);
  return inventory;
}

// ── GROUP BY SESSION ──────────────────────────
function groupBySessions(items) {
  const groups = {};
  const order = [];
  items.forEach(c => {
    if (!groups[c.session]) { groups[c.session]=[]; order.push(c.session); }
    groups[c.session].push(c);
  });
  return order.map(s=>({ session:s, cards:groups[s] }));
}

// ── UPDATE ASK ────────────────────────────────
function updateAsk(id, val) {
  const c = inventory.find(x=>x.id===id); if(!c) return;
  const parsed = parseFloat(val);
  c.ask = (!val || isNaN(parsed)) ? null : parsed;
  saveInv(inventory);
  // Update card border and profit display without full re-render
  const card = document.getElementById('invcard-'+id);
  const profEl = document.getElementById('profit-'+id);
  const pdProfit = document.getElementById('pdprofit-'+id);
  const pdMargin = document.getElementById('pdmargin-'+id);
  const p = profit(c);
  if (card) {
    card.classList.toggle('profitable', p!=null && p>0);
    card.classList.toggle('at-loss', p!=null && p<0);
  }
  if (profEl) {
    profEl.textContent = p!=null ? (p>=0?'+':'')+' $'+Math.abs(p).toFixed(2) : '—';
    profEl.className = 'inv-profit '+(p==null?'zero':p>0?'pos':p<0?'neg':'zero');
  }
  if (pdProfit) {
    pdProfit.textContent = p!=null ? (p>=0?'+':'')+' $'+Math.abs(p).toFixed(2) : '—';
    pdProfit.className = 'pd-val '+(p==null?'':p>0?'green':'danger');
  }
  if (pdMargin && c.ask!=null) {
    const m = ((c.ask-c.cost)/c.cost*100);
    pdMargin.textContent = (m>=0?'+':'')+m.toFixed(1)+'%';
    pdMargin.className = 'pd-val '+(m>=0?'green':'danger');
  } else if (pdMargin) { pdMargin.textContent='—'; pdMargin.className='pd-val'; }
  updateDashboard();
}

function updateDashboard() {
  const tc=document.getElementById('dashCost');   if(tc) tc.textContent='$'+totalCost().toFixed(2);
  const ta=document.getElementById('dashAsk');    if(ta) ta.textContent='$'+totalAsk().toFixed(2);
  const tp=document.getElementById('dashProfit'); if(tp) {
    const p=totalProfit();
    tp.textContent=(p>=0?'+':'')+' $'+Math.abs(p).toFixed(2);
    tp.className='dash-val '+(p>0?'green':p<0?'danger':'muted').replace('danger','').replace('muted','muted');
    if(p>0) tp.className='dash-val green'; else if(p<0) tp.className='dash-val danger'; else tp.className='dash-val muted';
  }
  const tm=document.getElementById('dashMargin'); if(tm) {
    const m=marginPct(); tm.textContent=(m>=0?'+':'')+m.toFixed(1)+'%';
  }
  const barFill=document.getElementById('profitBarFill');
  if(barFill) {
    const cost=inventory.filter(c=>c.ask!=null).reduce((s,c)=>s+c.cost,0);
    const ask=inventory.filter(c=>c.ask!=null).reduce((s,c)=>s+(c.ask||0),0);
    const pct = cost>0 ? Math.min(200, (ask/cost)*100) : 0;
    const fillPct = Math.min(100, Math.max(0, pct-100+50));
    barFill.style.width = fillPct+'%';
    barFill.classList.toggle('loss', totalProfit()<0);
  }
}

function toggleCard(id) {
  openId = openId===id ? null : id;
  const panel = document.getElementById('panel-'+id); if(!panel) return;
  panel.classList.toggle('open', openId===id);
  const card = document.getElementById('invcard-'+id); if(card) card.style.borderRadius = openId===id?'12px 12px 0 0':'12px';
  if (openId===id) setTimeout(()=>document.getElementById('ask-'+id)?.focus(), 80);
}

// ── RENDER ────────────────────────────────────
function render() {
  const body = document.getElementById('invBody');
  body.innerHTML = '';
  const items = filteredItems();

  // Dashboard
  const p = totalProfit();
  const m = marginPct();
  const cost = inventory.filter(c=>c.ask!=null).reduce((s,c)=>s+c.cost,0);
  const ask  = inventory.filter(c=>c.ask!=null).reduce((s,c)=>s+(c.ask||0),0);
  const barPct = cost>0 ? Math.min(100,Math.max(0,((ask/cost)*100)-100+50)) : 50;

  const dash = document.createElement('div');
  dash.className = 'dashboard';
  dash.innerHTML =
    '<div class="dash-card">'+
      '<div class="dash-lbl">INVESTED</div>'+
      '<div class="dash-val muted" id="dashCost">$'+totalCost().toFixed(2)+'</div>'+
      '<div class="dash-sub">'+inventory.length+' cards</div>'+
    '</div>'+
    '<div class="dash-card">'+
      '<div class="dash-lbl">HYPOTHETICAL RETURN</div>'+
      '<div class="dash-val accent" id="dashAsk">$'+totalAsk().toFixed(2)+'</div>'+
      '<div class="dash-sub">'+pricedCount()+' of '+inventory.length+' priced</div>'+
    '</div>'+
    '<div class="dash-card wide">'+
      '<div class="dash-lbl">HYPOTHETICAL PROFIT</div>'+
      '<div class="dash-val '+(p>0?'green':p<0?'danger':'muted')+'" id="dashProfit">'+(p>=0?'+':'')+' $'+Math.abs(p).toFixed(2)+'</div>'+
      '<div class="profit-bar-wrap">'+
        '<div class="profit-bar-track"><div class="profit-bar-fill'+(p<0?' loss':'')+'" id="profitBarFill" style="width:'+barPct+'%"></div></div>'+
        '<div class="profit-bar-labels"><span class="profit-bar-lbl">cost</span><span class="profit-bar-lbl" id="dashMargin">'+(m>=0?'+':'')+m.toFixed(1)+'%</span><span class="profit-bar-lbl">ask</span></div>'+
      '</div>'+
    '</div>';
  body.appendChild(dash);

  if (!items.length) {
    const empty = document.createElement('div');
    empty.className = 'empty-state';
    empty.innerHTML = '<div class="empty-icon">📦</div>'+(filter==='unpriced'?'All cards priced — nice.':'No inventory yet.\nComplete a sale to populate.');
    body.appendChild(empty);
    return;
  }

  // Group by session
  const groups = groupBySessions(items);
  groups.forEach(g => {
    const sessionCost = g.cards.reduce((s,c)=>s+c.cost,0);
    const sessionProfit = g.cards.filter(c=>c.ask!=null).reduce((s,c)=>s+(c.ask-c.cost),0);

    // Session header
    const sh = document.createElement('div');
    sh.className = 'session-header';
    sh.innerHTML =
      '<div class="session-title">'+g.session+'</div>'+
      '<div class="session-line"></div>'+
      '<div class="session-total">$'+sessionCost.toFixed(2)+(sessionProfit!==0?' · '+(sessionProfit>=0?'+':'')+' $'+Math.abs(sessionProfit).toFixed(2)+' est.':'')+'</div>';
    body.appendChild(sh);

    g.cards.forEach(c => {
      const p = profit(c);
      const card = document.createElement('div');
      card.className = 'inv-card'+(p!=null&&p>0?' profitable':p!=null&&p<0?' at-loss':'');
      card.id = 'invcard-'+c.id;

      const main = document.createElement('div');
      main.className = 'inv-main';
      main.onclick = ()=>toggleCard(c.id);

      const condPill = document.createElement('span');
      condPill.className = 'cond-pill cond-'+c.cond; condPill.textContent=c.cond;

      const left = document.createElement('div');
      left.className = 'inv-left';
      left.innerHTML =
        '<div class="inv-name">'+c.name+'</div>'+
        '<div class="inv-meta">'+
          '<span class="inv-number">'+c.number+'</span>'+
        '</div>';
      left.querySelector('.inv-meta').appendChild(condPill);

      const right = document.createElement('div');
      right.className = 'inv-right';
      right.innerHTML =
        '<div class="inv-cost">paid $'+c.cost.toFixed(2)+'</div>'+
        '<div class="inv-profit '+(p==null?'zero':p>0?'pos':'neg')+'" id="profit-'+c.id+'">'+(p!=null?(p>=0?'+':'')+' $'+Math.abs(p).toFixed(2):'set ask →')+'</div>';

      main.append(left, right);

      // Expand panel
      const panel = document.createElement('div');
      panel.className = 'inv-panel'+(openId===c.id?' open':'');
      panel.id = 'panel-'+c.id;

      const inner = document.createElement('div');
      inner.className = 'inv-panel-inner';

      // Ask price row
      const askRow = document.createElement('div');
      askRow.className = 'ask-row';
      askRow.innerHTML = '<div class="ask-label">ASKING PRICE</div>';
      const askInput = document.createElement('input');
      askInput.className = 'ask-input'; askInput.id = 'ask-'+c.id;
      askInput.type='number'; askInput.step='0.01'; askInput.min='0';
      askInput.placeholder='$0.00';
      if (c.ask!=null) askInput.value = c.ask.toFixed(2);
      askInput.oninput = ()=>updateAsk(c.id, askInput.value);
      askRow.appendChild(askInput);

      // Profit breakdown
      const pd = document.createElement('div');
      pd.className = 'profit-detail';
      const pVal = profit(c);
      const mVal = c.ask!=null ? ((c.ask-c.cost)/c.cost*100) : null;
      pd.innerHTML =
        '<div class="profit-detail-item"><div class="pd-lbl">PAID</div><div class="pd-val">$'+c.cost.toFixed(2)+'</div></div>'+
        '<div class="pd-arrow">→</div>'+
        '<div class="profit-detail-item"><div class="pd-lbl">ASK</div><div class="pd-val">'+( c.ask!=null?'$'+c.ask.toFixed(2):'—' )+'</div></div>'+
        '<div class="pd-arrow">=</div>'+
        '<div class="profit-detail-item"><div class="pd-lbl">PROFIT</div><div class="pd-val '+(pVal==null?'':pVal>=0?'green':'danger')+'" id="pdprofit-'+c.id+'">'+(pVal!=null?(pVal>=0?'+':'')+' $'+Math.abs(pVal).toFixed(2):'—')+'</div></div>'+
        '<div class="profit-detail-item"><div class="pd-lbl">MARGIN</div><div class="pd-val '+(mVal==null?'':mVal>=0?'green':'danger')+'" id="pdmargin-'+c.id+'">'+(mVal!=null?(mVal>=0?'+':'')+mVal.toFixed(1)+'%':'—')+'</div></div>';

      inner.append(askRow, pd);
      panel.appendChild(inner);
      card.append(main, panel);
      body.appendChild(card);
    });
  });
}

function el(tag,cls,text){
  const e=document.createElement(tag); if(cls) e.className=cls; if(text) e.textContent=text; return e;
}

render();
